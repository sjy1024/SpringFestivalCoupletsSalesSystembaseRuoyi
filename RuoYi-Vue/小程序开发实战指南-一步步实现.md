## 📝 第2步：配置请求工具和API

### 2.1 创建请求封装工具

**位置：** `uni-app-miniprogram/utils/request.js`

**内容：**

```javascript
// 请求基础URL
const BASE_URL = 'http://localhost:8080'

/**
 * 请求封装
 */
const request = (options) => {
  return new Promise((resolve, reject) => {
    // 获取 deviceId
    const deviceId = uni.getStorageSync('deviceId') || ''
    
    uni.request({
      url: BASE_URL + options.url,
      method: options.method || 'GET',
      data: options.data || {},
      header: {
        'Content-Type': 'application/json',
        'deviceId': deviceId,  // 添加 deviceId 到请求头
        ...options.header
      },
      success: (res) => {
        // 统一处理响应
        if (res.statusCode === 200) {
          if (res.data.code === 200) {
            resolve(res.data)
          } else {
            uni.showToast({
              title: res.data.msg || '请求失败',
              icon: 'none'
            })
            reject(res.data)
          }
        } else {
          uni.showToast({
            title: '网络错误',
            icon: 'none'
          })
          reject(res)
        }
      },
      fail: (error) => {
        uni.showToast({
          title: '请求失败',
          icon: 'none'
        })
        reject(error)
      }
    })
  })
}

export default request
```

### 2.2 创建 deviceId 管理工具

**位置：** `uni-app-miniprogram/utils/device.js`

**内容：**

```javascript
/**
 * 获取或生成 deviceId
 */
export function getDeviceId() {
  let deviceId = uni.getStorageSync('deviceId')
  
  if (!deviceId) {
    // 生成 deviceId（时间戳 + 随机数）
    deviceId = 'device_' + Date.now() + '_' + Math.random().toString(36).substr(2, 9)
    uni.setStorageSync('deviceId', deviceId)
  }
  
  return deviceId
}

/**
 * 清除 deviceId
 */
export function clearDeviceId() {
  uni.removeStorageSync('deviceId')
}
```

```javascript
  // 商品列表
  list: (params) => {
```

```vue
<template>
  <view class="container">
    <!-- 商品列表 -->
    <view class="goods-list">
      <view 
        class="goods-item" 
        v-for="item in goodsList" 
        :key="item.goodsId"
        @click="goDetail(item.goodsId)"
      >
        <image :src="item.coverImage" class="goods-image" mode="aspectFill"></image>
        <view class="goods-info">
          <text class="goods-name">{{ item.goodsName }}</text>
          <text class="goods-price">¥{{ item.goodsPrice }}</text>
          <text class="goods-stock">库存：{{ item.stock }}</text>
        </view>
      </view>
    </view>
    
    <!-- 空状态 -->
    <view v-if="goodsList.length === 0 && !loading" class="empty">
      <text>暂无商品</text>
    </view>
    
    <!-- 加载中 -->
    <view v-if="loading" class="loading">
      <text>加载中...</text>
    </view>
  </view>
</template>

<script>
import { goodsApi } from '@/utils/api.js'

export default {
  data() {
    return {
      goodsList: [],
      loading: false
    }
  },
  onLoad() {
    this.loadGoodsList()
  },
  // 下拉刷新
  onPullDownRefresh() {
    this.loadGoodsList().then(() => {
      uni.stopPullDownRefresh()
    })
  },
  methods: {
    async loadGoodsList() {
      this.loading = true
      try {
        const res = await goodsApi.list({
          status: '0'  // 只查询上架的商品
        })
        if (res.code === 200) {
          this.goodsList = res.data || res.rows || []
        }
      } catch (error) {
        console.error('加载商品列表失败', error)
      } finally {
        this.loading = false
      }
    },
    goDetail(goodsId) {
      uni.navigateTo({
        url: `/pages/goods/detail?id=${goodsId}`
      })
    }
  }
}
</script>
```

## 📝 第5步：实现下单功能

### 5.1 创建下单页面

**位置：** `uni-app-miniprogram/pages/order/create.vue`

**内容：**

```vue
<template>
  <view class="container">
    <!-- 收货信息 -->
    <view class="section">
      <view class="section-title">收货信息</view>
      <view class="form-item">
        <text class="label">收货人：</text>
        <input v-model="receiverName" placeholder="请输入收货人" class="input" />
      </view>
      <view class="form-item">
        <text class="label">联系电话：</text>
        <input v-model="receiverPhone" placeholder="请输入联系电话" type="number" class="input" />
      </view>
      <view class="form-item">
        <text class="label">收货地址：</text>
        <textarea v-model="receiverAddress" placeholder="请输入收货地址" class="textarea" />
      </view>
    </view>
    
    <!-- 商品信息 -->
    <view class="section">
      <view class="section-title">商品信息</view>
      <view class="goods-item">
        <image :src="goodsInfo.coverImage" class="goods-image" mode="aspectFill"></image>
        <view class="goods-info">
          <text class="goods-name">{{ goodsInfo.goodsName }}</text>
          <text class="goods-price">¥{{ goodsInfo.goodsPrice }} × {{ quantity }}</text>
        </view>
      </view>
    </view>
    
    <!-- 订单金额 -->
    <view class="section">
      <view class="amount-item">
        <text class="label">商品金额：</text>
        <text class="value">¥{{ totalAmount }}</text>
      </view>
      <view class="amount-item total">
        <text class="label">订单总额：</text>
        <text class="value">¥{{ totalAmount }}</text>
      </view>
    </view>
    
    <!-- 提交订单按钮 -->
    <view class="footer">
      <button @click="submitOrder" :disabled="submitting" class="submit-btn">
        {{ submitting ? '提交中...' : '提交订单' }}
      </button>
    </view>
  </view>
</template>

<script>
import { goodsApi, orderApi } from '@/utils/api.js'
import { getDeviceId } from '@/utils/device.js'

export default {
  data() {
    return {
      goodsId: '',
      quantity: 1,
      goodsInfo: {},
      receiverName: '',
      receiverPhone: '',
      receiverAddress: '',
      submitting: false
    }
  },
  computed: {
    totalAmount() {
      if (this.goodsInfo.goodsPrice && this.quantity) {
        return (parseFloat(this.goodsInfo.goodsPrice) * this.quantity).toFixed(2)
      }
      return '0.00'
    }
  },
  onLoad(options) {
    this.goodsId = options.goodsId
    this.quantity = parseInt(options.quantity) || 1
    this.loadGoodsDetail()
  },
  methods: {
    async loadGoodsDetail() {
      try {
        const res = await goodsApi.detail(this.goodsId)
        if (res.code === 200) {
          this.goodsInfo = res.data
        }
      } catch (error) {
        console.error('加载商品详情失败', error)
      }
    },
    async submitOrder() {
      // 验证表单
      if (!this.receiverName) {
        uni.showToast({
          title: '请输入收货人',
          icon: 'none'
        })
        return
      }
      if (!this.receiverPhone) {
        uni.showToast({
          title: '请输入联系电话',
          icon: 'none'
        })
        return
      }
      if (!this.receiverAddress) {
        uni.showToast({
          title: '请输入收货地址',
          icon: 'none'
        })
        return
      }
      
      this.submitting = true
      try {
        const res = await orderApi.create({
          goodsId: this.goodsId,
          quantity: this.quantity,
          receiverName: this.receiverName,
          receiverPhone: this.receiverPhone,
          receiverAddress: this.receiverAddress,
          deviceId: getDeviceId()
        })
        
        if (res.code === 200) {
          uni.showToast({
            title: '下单成功',
            icon: 'success'
          })
          // 跳转到订单列表
          setTimeout(() => {
            uni.switchTab({
              url: '/pages/order/list'
            })
          }, 1500)
        }
      } catch (error) {
        console.error('下单失败', error)
      } finally {
        this.submitting = false
      }
    }
  }
}
</script>
```

```vue
import { orderApi } from '@/utils/api.js'
import { getDeviceId } from '@/utils/device.js'

    }

    this.loadOrderList()
  },

    // 每次显示时刷新列表

  },

    this.loadOrderList().then(() => {
      uni.stopPullDownRefresh()
    })

  methods: {
    async loadOrderList() {
      this.loading = true
      try {
        const res = await orderApi.list({
          deviceId: getDeviceId()
        })
        if (res.code === 200) {
          this.orderList = res.data || res.rows || []
        }
      } catch (error) {
        console.error('加载订单列表失败', error)
      } finally {
        this.loading = false
      }
    },
    getStatusText(status) {
      const statusMap = {

        '1': '已支付',
        '2': '制作中',
        '3': '已完成',
        '4': '已取消',
        '5': '已收货'
      }
      return statusMap[status] || '未知'

      try {
        const res = await orderApi.mockPay(orderId)
        if (res.code === 200) {
          uni.showToast({
            title: '支付成功',
            icon: 'success'
          })
          this.loadOrderList()
        }
      } catch (error) {
        console.error('支付失败', error)
      }
    },
    async confirmReceive(orderId) {

              title: '确认收货成功',
              icon: 'success'
            })
            this.loadOrderList()
          }
        }
      })
```
